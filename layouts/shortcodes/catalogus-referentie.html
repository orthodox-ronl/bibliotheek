{{- /* Boom-overzichten over de catalogus (build-time). kind: alfabetisch | herkomst | per-soort alfabetisch / per-soort: zangstuk → [variant] → blad herkomst: uitvoeringsvorm-id → zangstuk → [variant] → blad Enige variant ``default``/``standaard``: variant-tak wordt overgeslagen; bladtitel is ``Zangstuk (uitvoeringsvorm)`` zonder «Standaard». */ -}} {{- $kind := .Get "kind" | default "alfabetisch" -}} {{- $loc := partial "diag-location.html" (dict "position" .Position "page" .Page) -}} {{- $bib := site.GetPage "/catalogus" -}} {{- if not $bib -}} {{- errorf "%s: shortcode catalogus-referentie: sectie catalogus ontbreekt. "id" $id "title" $title "titleSort" (lower $title) "z" $z "v" $v "u" $u "zLabel" $zLabel "vLabel" $vLabel "uLabel" $uLabel "zSort" (lower $zLabel) "vSort" (lower $vLabel) "uSort" (lower $uLabel) "zWeight" $zw ) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Guard: default/standaard mag niet naast andere varianten bestaan. */ -}} {{- $zAll := slice -}} {{- $zSeenGuard := slice -}} {{- range $leaves -}} {{- if not (in $zSeenGuard .z) -}} {{- $zSeenGuard = $zSeenGuard | append .z -}} {{- $zAll = $zAll | append .z -}} {{- end -}} {{- end -}} {{- range $zAll -}} {{- $zKey := . -}} {{- $zLeaves := where $leaves "z" $zKey -}} {{- $hasDefault := false -}} {{- $hasOther := false -}} {{- $otherList := slice -}} {{- range $zLeaves -}} {{- if or (eq .v "default") (eq .v "standaard") -}} {{- $hasDefault = true -}} {{- else -}} {{- $hasOther = true -}} {{- if not (in $otherList .v) -}} {{- $otherList = $otherList | append .v -}} {{- end -}} {{- end -}} {{- end -}} {{- if and $hasDefault $hasOther -}} {{- errorf "%s: zangstuk %q heeft variant default/standaard én andere varianten (%s). Dat mag niet; kies één model. Oplossing: hernoem of consolideer varianten in de catalogus." $loc $zKey (delimit $otherList ", ") -}} {{- end -}} {{- end -}} {{- if not $leaves -}}
