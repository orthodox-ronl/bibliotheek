{{- /* Blad-PDF-URL (docs/plans/pdf-r2.md): lokaal-eerst, anders R2. Context dict: page — pagina met .File local — relatieve site-URL als het bestand lokaal/in de build zit (mag leeg) filename — bijv. index.md.pdf params.pdfAssets: base — publieke R2-host (eindigend op /) preferLocal — true (default): local wint; false: altijd R2 (main-build) allowDevelopmentOverrides — true op preview: pad in data/pdf-r2-development.json → development/, anders production/ */ -}} {{- $page := .page -}} {{- $local := .local | default "" -}} {{- $filename := .filename -}} {{- $cfg := $page.Site.Params.pdfAssets | default dict -}} {{- $base := strings.TrimSpace ($cfg.base | default "") -}} {{- $preferLocal := true -}} {{- if isset $cfg "preferlocal" -}} {{- $preferLocal = $cfg.preferlocal -}} {{- end -}} {{- $allowDev := false -}} {{- if isset $cfg "allowdevelopmentoverrides" -}} {{- $allowDev = $cfg.allowdevelopmentoverrides -}} {{- end -}} {{- $r2Env := "production" -}} {{- $r2 := "" -}} {{- if and (ne $base "") $page.File -}} {{- $dir := strings.TrimSuffix "/" (replace $page.File.Dir "\\" "/") -}} {{- $rel := "" -}} {{- if $dir -}} {{- $rel = printf "%s/%s" $dir $filename -}} {{- else -}} {{- $rel = $filename -}} {{- end -}} {{- if $allowDev -}} {{- $man := index hugo.Data "pdf-r2-development" | default dict -}} {{- $overrides := $man.overrides | default slice -}} {{- range $overrides -}} {{- if eq . $rel -}} {{- $r2Env = "development" -}} {{- end -}} {{- end -}} {{- end -}} {{- $base = strings.TrimSuffix "/" $base -}} {{- $r2 = printf "%s/%s/%s" $base $r2Env $rel -}} {{- end -}} {{- if and $preferLocal (ne $local "") -}} {{- $local -}} {{- else if ne $r2 "" -}} {{- $r2 -}} {{- else -}} {{- $local -}} {{- end -}}
