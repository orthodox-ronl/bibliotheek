{{- /* Variant-laag onder een zangstuk → nodes voor catalogus-boom-nodes. Enige variant (elke id, dus ook default) wordt door single-child inklappen weggelaten; zie catalogus-boom-nodes.html. Guard default+andere blijft in catalogus-referentie. Context: dict leaves (slice leaf-dicts met v/vLabel/vSort/leafSort). */ -}} {{- $leaves := .leaves -}} {{- $vSeen := slice -}} {{- $vNodes := slice -}} {{- range $leaves -}} {{- if not (in $vSeen .v) -}} {{- $vSeen = $vSeen | append .v -}} {{- $vNodes = $vNodes | append (dict "key" .v "label" .vLabel "sort" .vSort) -}} {{- end -}} {{- end -}} {{- $vNodes = sort $vNodes "sort" -}} {{- $nodes := slice -}} {{- range $vNodes -}} {{- $vKey := .key -}} {{- $vLeaves := slice -}} {{- range $leaves -}} {{- if eq .v $vKey -}} {{- $vLeaves = $vLeaves | append . -}} {{- end -}} {{- end -}} {{- $vLeaves = sort $vLeaves "leafSort" -}} {{- $children := slice -}} {{- range $vLeaves -}} {{- $children = $children | append (dict "kind" "leaf" "leaf" .) -}} {{- end -}} {{- $nodes = $nodes | append (dict "kind" "branch" "label" .label "children" $children "leafCount" (len $vLeaves) ) -}} {{- end -}} {{- partial "catalogus-boom-nodes.html" (dict "nodes" $nodes) -}}
